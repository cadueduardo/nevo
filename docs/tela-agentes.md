export const ACTIVE_AGENT_KEY = 'nevo_active_agent_id';

export function getActiveAgentId(): string | null {
  if (typeof window === 'undefined') return null;
  return window.localStorage.getItem(ACTIVE_AGENT_KEY);
}

export function setActiveAgentId(agentId: string) {
  if (typeof window === 'undefined') return;
  window.localStorage.setItem(ACTIVE_AGENT_KEY, agentId);
}

export function clearActiveAgentId() {
  if (typeof window === 'undefined') return;
  window.localStorage.removeItem(ACTIVE_AGENT_KEY);
}

'use client';

import * as React from 'react';
import { mockAgents, type Agent } from '@/lib/mock/agents';
import { getActiveAgentId, setActiveAgentId as persistActiveAgentId } from '@/lib/agents/active-agent';

type AgentContextValue = {
  agents: Agent[];
  activeAgent: Agent | null;
  activeAgentId: string | null;
  setActiveAgentId: (id: string) => void;
  // event bus simple
  notifyAgentConfigUpdated: (reason?: string) => void;
  lastConfigUpdateAt: number | null;
  lastConfigUpdateReason?: string;
};

const AgentContext = React.createContext<AgentContextValue | null>(null);

export function useAgentContext() {
  const ctx = React.useContext(AgentContext);
  if (!ctx) throw new Error('useAgentContext must be used within AgentProvider');
  return ctx;
}

export function AgentProvider({ children }: { children: React.ReactNode }) {
  const [agents] = React.useState<Agent[]>(mockAgents);
  const [activeAgentId, _setActiveAgentId] = React.useState<string | null>(null);
  const [lastConfigUpdateAt, setLastConfigUpdateAt] = React.useState<number | null>(null);
  const [lastConfigUpdateReason, setLastConfigUpdateReason] = React.useState<string | undefined>(undefined);

  React.useEffect(() => {
    const stored = getActiveAgentId();
    if (stored && agents.some(a => a.id === stored)) _setActiveAgentId(stored);
    else if (agents[0]) {
      _setActiveAgentId(agents[0].id);
      persistActiveAgentId(agents[0].id);
    }
  }, [agents]);

  const activeAgent = React.useMemo(
    () => agents.find(a => a.id === activeAgentId) ?? null,
    [agents, activeAgentId]
  );

  const setActiveAgentId = React.useCallback((id: string) => {
    _setActiveAgentId(id);
    persistActiveAgentId(id);
  }, []);

  const notifyAgentConfigUpdated = React.useCallback((reason?: string) => {
    setLastConfigUpdateAt(Date.now());
    setLastConfigUpdateReason(reason);
    // Optional: dispatch a DOM event too, for non-react listeners
    if (typeof window !== 'undefined') {
      window.dispatchEvent(new CustomEvent('nevo:agent_config_updated', { detail: { reason } }));
    }
  }, []);

  return (
    <AgentContext.Provider
      value={{
        agents,
        activeAgent,
        activeAgentId,
        setActiveAgentId,
        notifyAgentConfigUpdated,
        lastConfigUpdateAt,
        lastConfigUpdateReason,
      }}
    >
      {children}
    </AgentContext.Provider>
  );
}

/* Inspector:
   - Para nós message/question: editar texto e tipo de UI (buttons/list/text) SEM quebrar WhatsApp
   - Para IA: editar prompt template, schema de saída (placeholder), mostrar variáveis disponíveis
   - Para condition: editar branches e labels
   - Sempre mostrar “Compatibilidade WhatsApp” (validação visual)
*/
function FlowInspector({ node }: { node: FlowNode | null }) {
  if (!node) {
    return (
      <Card>
        <CardContent className="p-4 text-sm text-muted-foreground">
          Selecione um nó para editar.
        </CardContent>
      </Card>
    );
  }

  return (
    <Card>
      <CardContent className="p-4">
        <div className="flex items-start justify-between gap-2">
          <div className="min-w-0">
            <div className="truncate font-semibold">{node.title}</div>
            <div className="mt-1"><NodeTypeBadge type={node.type} /></div>
          </div>
          <Button size="sm">Salvar</Button>
        </div>

        <div className="mt-4 space-y-3 text-sm">
          <div className="rounded-lg border p-3">
            <div className="font-medium">Compatibilidade WhatsApp</div>
            <div className="mt-1 text-xs text-muted-foreground">
              Este nó deve gerar apenas componentes suportados (texto, botões, lista). Nada fora disso.
            </div>
          </div>

          {node.type === 'ai' ? (
            <div className="rounded-lg border p-3">
              <div className="font-medium">Prompt</div>
              <div className="mt-1 text-xs text-muted-foreground">
                Mostrar editor de prompt (textarea), variáveis disponíveis e schema de saída.
              </div>
              <div className="mt-2 rounded-md border bg-muted/30 p-2 text-xs">
                {node.data?.promptPreview ?? '—'}
              </div>
            </div>
          ) : null}

          {(node.type === 'message' || node.type === 'question') ? (
            <div className="rounded-lg border p-3">
              <div className="font-medium">Mensagem</div>
              <div className="mt-1 text-xs text-muted-foreground">
                Editor + preview WhatsApp ao vivo.
              </div>
              <div className="mt-2">
                <WhatsAppPreview message={node.data?.message ?? ''} ui={node.data?.ui} />
              </div>
            </div>
          ) : null}

          {node.type === 'condition' ? (
            <div className="rounded-lg border p-3">
              <div className="font-medium">Branches</div>
              <div className="mt-1 text-xs text-muted-foreground">
                Editar opções (ex.: booking/quote/other) e labels das saídas.
              </div>
            </div>
          ) : null}
        </div>
      </CardContent>
    </Card>
  );
}
